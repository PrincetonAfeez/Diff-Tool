#   pip install pip-tools
#   pip-compile --extra=dev --output-file=requirements-dev.lock pyproject.toml
#
# Install:
#   pip install -e .
#   pip install -r requirements-dev.lock
#
# This file delegates to the lockfile so a single command still works:
